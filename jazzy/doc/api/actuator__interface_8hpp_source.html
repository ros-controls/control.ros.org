165 CallbackReturn init(const hardware_interface::HardwareComponentParams & params)
166 {
167 actuator_clock_ = params.clock;
168 auto logger_copy = params.logger;
169 actuator_logger_ =
170 logger_copy.get_child("hardware_component.actuator." + params.hardware_info.name);
171 info_ = params.hardware_info;
172 if (info_.is_async)
173 {
174 RCLCPP_INFO_STREAM(
175 get_logger(), "Starting async handler with scheduler priority: " << info_.thread_priority);
176 async_handler_ = std::make_unique<realtime_tools::AsyncFunctionHandler<return_type>>();
177 async_handler_->init(
178 [this](const rclcpp::Time & time, const rclcpp::Duration & period)
179 {
180 const auto read_start_time = std::chrono::steady_clock::now();
181 const auto ret_read = read(time, period);
182 const auto read_end_time = std::chrono::steady_clock::now();
183 read_return_info_.store(ret_read, std::memory_order_release);
184 read_execution_time_.store(
185 std::chrono::duration_cast<std::chrono::nanoseconds>(read_end_time - read_start_time),
186 std::memory_order_release);
187 if (ret_read != return_type::OK)
188 {
189 return ret_read;
190 }
191 const auto write_start_time = std::chrono::steady_clock::now();
192 const auto ret_write = write(time, period);
193 const auto write_end_time = std::chrono::steady_clock::now();
194 write_return_info_.store(ret_write, std::memory_order_release);
195 write_execution_time_.store(
196 std::chrono::duration_cast<std::chrono::nanoseconds>(write_end_time - write_start_time),
197 std::memory_order_release);
198 return ret_write;
199 },
200 info_.thread_priority);
201 async_handler_->start_thread();
202 }
203 hardware_interface::HardwareComponentInterfaceParams interface_params;
204 interface_params.hardware_info = info_;
205 interface_params.executor = params.executor;
206 return on_init(interface_params);
207 };

284 virtual std::vector<StateInterface::ConstSharedPtr> on_export_state_interfaces()
285 {
286 // import the unlisted interfaces
287 std::vector<hardware_interface::InterfaceDescription> unlisted_interface_descriptions =
288 export_unlisted_state_interface_descriptions();
289
290 std::vector<StateInterface::ConstSharedPtr> state_interfaces;
291 state_interfaces.reserve(
292 unlisted_interface_descriptions.size() + joint_state_interfaces_.size());
293
294 // add InterfaceDescriptions and create the StateInterfaces from the descriptions and add to
295 // maps.
296 for (const auto & description : unlisted_interface_descriptions)
297 {
298 auto name = description.get_name();
299 unlisted_state_interfaces_.insert(std::make_pair(name, description));
300 auto state_interface = std::make_shared<StateInterface>(description);
301 actuator_states_.insert(std::make_pair(name, state_interface));
302 unlisted_states_.push_back(state_interface);
303 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
304 }
305
306 for (const auto & [name, descr] : joint_state_interfaces_)
307 {
308 auto state_interface = std::make_shared<StateInterface>(descr);
309 actuator_states_.insert(std::make_pair(name, state_interface));
310 joint_states_.push_back(state_interface);
311 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
312 }
313 return state_interfaces;
314 }

439 HardwareComponentCycleStatus trigger_read(
440 const rclcpp::Time & time, const rclcpp::Duration & period)
441 {
442 HardwareComponentCycleStatus status;
443 status.result = return_type::ERROR;
444 if (info_.is_async)
445 {
446 status.result = read_return_info_.load(std::memory_order_acquire);
447 const auto read_exec_time = read_execution_time_.load(std::memory_order_acquire);
448 if (read_exec_time.count() > 0)
449 {
450 status.execution_time = read_exec_time;
451 }
452 const auto result = async_handler_->trigger_async_callback(time, period);
453 status.successful = result.first;
454 if (!status.successful)
455 {
456 RCLCPP_WARN(
457 get_logger(),
458 "Trigger read/write called while the previous async trigger is still in progress for "
459 "hardware interface : '%s'. Failed to trigger read/write cycle!",
460 info_.name.c_str());
461 status.result = return_type::OK;
462 return status;
463 }
464 }
465 else
466 {
467 const auto start_time = std::chrono::steady_clock::now();
468 status.successful = true;
469 status.result = read(time, period);
470 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
471 std::chrono::steady_clock::now() - start_time);
472 }
473 return status;
474 }

498 HardwareComponentCycleStatus trigger_write(
499 const rclcpp::Time & time, const rclcpp::Duration & period)
500 {
501 HardwareComponentCycleStatus status;
502 status.result = return_type::ERROR;
503 if (info_.is_async)
504 {
505 status.successful = true;
506 const auto write_exec_time = write_execution_time_.load(std::memory_order_acquire);
507 if (write_exec_time.count() > 0)
508 {
509 status.execution_time = write_exec_time;
510 }
511 status.result = write_return_info_.load(std::memory_order_acquire);
512 }
513 else
514 {
515 const auto start_time = std::chrono::steady_clock::now();
516 status.successful = true;
517 status.result = write(time, period);
518 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
519 std::chrono::steady_clock::now() - start_time);
520 }
521 return status;
522 }

562 template <typename T>
563 void set_state(const std::string & interface_name, const T & value)
564 {
565 auto it = actuator_states_.find(interface_name);
566 if (it == actuator_states_.end())
567 {
568 throw std::runtime_error(
569 fmt::format(
570 FMT_COMPILE(
571 "State interface not found: {} in actuator hardware component: {}. "
572 "This should not happen."),
573 interface_name, info_.name));
574 }
575 auto & handle = it->second;
576 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
577 std::ignore = handle->set_value(lock, value);
578 }

580 template <typename T = double>
581 T get_state(const std::string & interface_name) const
582 {
583 auto it = actuator_states_.find(interface_name);
584 if (it == actuator_states_.end())
585 {
586 throw std::runtime_error(
587 fmt::format(
588 FMT_COMPILE(
589 "State interface not found: {} in actuator hardware component: {}. "
590 "This should not happen."),
591 interface_name, info_.name));
592 }
593 auto & handle = it->second;
594 std::shared_lock<std::shared_mutex> lock(handle->get_mutex());
595 const auto opt_value = handle->get_optional<T>(lock);
596 if (!opt_value)
597 {
598 throw std::runtime_error(
599 fmt::format(
600 FMT_COMPILE("Failed to get state value from interface: {}. This should not happen."),
601 interface_name));
602 }
603 return opt_value.value();
