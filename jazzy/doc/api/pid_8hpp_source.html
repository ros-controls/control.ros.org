47class AntiwindupStrategy
48{
49public:
50 enum Value : int8_t
51 {
52 NONE = 0,
53 BACK_CALCULATION,
54 CONDITIONAL_INTEGRATION
55 };
56
57 constexpr AntiwindupStrategy() : value_(NONE) {}
58 constexpr AntiwindupStrategy(Value v) : value_(v) {} // NOLINT(runtime/explicit)
59
60 explicit AntiwindupStrategy(const std::string & s)
61 {
62 if (s == "back_calculation")
63 {
64 value_ = BACK_CALCULATION;
65 }
66 else if (s == "conditional_integration")
67 {
68 value_ = CONDITIONAL_INTEGRATION;
69 }
70 else
71 {
72 value_ = NONE;
73 }
74 }
75
76 operator std::string() const { return to_string(); }
77
78 constexpr bool operator==(AntiwindupStrategy other) const { return value_ == other.value_; }
79 constexpr bool operator!=(AntiwindupStrategy other) const { return value_ != other.value_; }
80
81 constexpr bool operator==(Value other) const { return value_ == other; }
82 constexpr bool operator!=(Value other) const { return value_ != other; }
83
84 std::string to_string() const
85 {
86 switch (value_)
87 {
88 case BACK_CALCULATION:
89 return "back_calculation";
90 case CONDITIONAL_INTEGRATION:
91 return "conditional_integration";
92 case NONE:
93 default:
94 return "none";
95 }
96 }
97
98private:
99 Value value_;
100};
