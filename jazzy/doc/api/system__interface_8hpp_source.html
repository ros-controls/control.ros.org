169 CallbackReturn init(const hardware_interface::HardwareComponentParams & params)
170 {
171 system_clock_ = params.clock;
172 auto logger_copy = params.logger;
173 system_logger_ =
174 logger_copy.get_child("hardware_component.system." + params.hardware_info.name);
175 info_ = params.hardware_info;
176 if (info_.is_async)
177 {
178 RCLCPP_INFO_STREAM(
179 get_logger(), "Starting async handler with scheduler priority: " << info_.thread_priority);
180 async_handler_ = std::make_unique<realtime_tools::AsyncFunctionHandler<return_type>>();
181 async_handler_->init(
182 [this](const rclcpp::Time & time, const rclcpp::Duration & period)
183 {
184 const auto read_start_time = std::chrono::steady_clock::now();
185 const auto ret_read = read(time, period);
186 const auto read_end_time = std::chrono::steady_clock::now();
187 read_return_info_.store(ret_read, std::memory_order_release);
188 read_execution_time_.store(
189 std::chrono::duration_cast<std::chrono::nanoseconds>(read_end_time - read_start_time),
190 std::memory_order_release);
191 if (ret_read != return_type::OK)
192 {
193 return ret_read;
194 }
195 const auto write_start_time = std::chrono::steady_clock::now();
196 const auto ret_write = write(time, period);
197 const auto write_end_time = std::chrono::steady_clock::now();
198 write_return_info_.store(ret_write, std::memory_order_release);
199 write_execution_time_.store(
200 std::chrono::duration_cast<std::chrono::nanoseconds>(write_end_time - write_start_time),
201 std::memory_order_release);
202 return ret_write;
203 },
204 info_.thread_priority);
205 async_handler_->start_thread();
206 }
207
208 if (auto locked_executor = params.executor.lock())
209 {
210 std::string node_name = params.hardware_info.name;
211 std::transform(
212 node_name.begin(), node_name.end(), node_name.begin(),
213 [](unsigned char c) { return std::tolower(c); });
214 std::replace(node_name.begin(), node_name.end(), '/', '_');
215 hardware_component_node_ = std::make_shared<rclcpp::Node>(node_name);
216 locked_executor->add_node(hardware_component_node_->get_node_base_interface());
217 }
218 else
219 {
220 RCLCPP_WARN(
221 params.logger,
222 "Executor is not available during hardware component initialization for '%s'. Skipping "
223 "node creation!",
224 params.hardware_info.name.c_str());
225 }
226
227 hardware_interface::HardwareComponentInterfaceParams interface_params;
228 interface_params.hardware_info = info_;
229 interface_params.executor = params.executor;
230 return on_init(interface_params);
231 };

314 virtual std::vector<StateInterface::ConstSharedPtr> on_export_state_interfaces()

316 // import the unlisted interfaces

318 export_unlisted_state_interface_descriptions();

322 unlisted_interface_descriptions.size() + joint_state_interfaces_.size() +
323 sensor_state_interfaces_.size() + gpio_state_interfaces_.size());
324
325 // add InterfaceDescriptions and create the StateInterfaces from the descriptions and add to
326 // maps.

329 auto name = description.get_name();
330 unlisted_state_interfaces_.insert(std::make_pair(name, description));
331 auto state_interface = std::make_shared<StateInterface>(description);
332 system_states_.insert(std::make_pair(name, state_interface));
333 unlisted_states_.push_back(state_interface);
334 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
335 }
336
337 for (const auto & [name, descr] : joint_state_interfaces_)
338 {
339 auto state_interface = std::make_shared<StateInterface>(descr);
340 system_states_.insert(std::make_pair(name, state_interface));
341 joint_states_.push_back(state_interface);
342 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
343 }
344 for (const auto & [name, descr] : sensor_state_interfaces_)

346 auto state_interface = std::make_shared<StateInterface>(descr);

491 HardwareComponentCycleStatus trigger_read(
492 const rclcpp::Time & time, const rclcpp::Duration & period)
493 {
494 HardwareComponentCycleStatus status;
495 status.result = return_type::ERROR;
496 if (info_.is_async)
497 {
498 status.result = read_return_info_.load(std::memory_order_acquire);
499 const auto read_exec_time = read_execution_time_.load(std::memory_order_acquire);
500 if (read_exec_time.count() > 0)
501 {
502 status.execution_time = read_exec_time;
503 }
504 const auto result = async_handler_->trigger_async_callback(time, period);
505 status.successful = result.first;
506 if (!status.successful)
507 {
508 RCLCPP_WARN(
509 get_logger(),
510 "Trigger read/write called while the previous async trigger is still in progress for "
511 "hardware interface : '%s'. Failed to trigger read/write cycle!",
512 info_.name.c_str());
513 status.result = return_type::OK;
514 return status;
515 }
516 }
517 else
518 {
519 const auto start_time = std::chrono::steady_clock::now();
520 status.successful = true;
521 status.result = read(time, period);
522 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
523 std::chrono::steady_clock::now() - start_time);
524 }
525 return status;
526 }

550 HardwareComponentCycleStatus trigger_write(
551 const rclcpp::Time & time, const rclcpp::Duration & period)
552 {
553 HardwareComponentCycleStatus status;
554 status.result = return_type::ERROR;
555 if (info_.is_async)
556 {
557 status.successful = true;
558 const auto write_exec_time = write_execution_time_.load(std::memory_order_acquire);
559 if (write_exec_time.count() > 0)
560 {
561 status.execution_time = write_exec_time;
562 }
563 status.result = write_return_info_.load(std::memory_order_acquire);
564 }
565 else
566 {
567 const auto start_time = std::chrono::steady_clock::now();
568 status.successful = true;
569 status.result = write(time, period);
570 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
571 std::chrono::steady_clock::now() - start_time);
572 }
573 return status;

614 template <typename T>
615 void set_state(const std::string & interface_name, const T & value)
616 {
617 auto it = system_states_.find(interface_name);
618 if (it == system_states_.end())
619 {
620 throw std::runtime_error(
621 fmt::format(
622 FMT_COMPILE(
623 "State interface not found: {} in system hardware component: {}. "
624 "This should not happen."),
625 interface_name, info_.name));
626 }
627 auto & handle = it->second;
628 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
629 std::ignore = handle->set_value(lock, value);
630 }

632 template <typename T = double>
633 T get_state(const std::string & interface_name) const
634 {
635 auto it = system_states_.find(interface_name);
636 if (it == system_states_.end())

644 }
645 auto & handle = it->second;
646 std::shared_lock<std::shared_mutex> lock(handle->get_mutex());
647 const auto opt_value = handle->get_optional<T>(lock);
648 if (!opt_value)
649 {
650 throw std::runtime_error(
651 fmt::format(
652 FMT_COMPILE("Failed to get state value from interface: {}. This should not happen."),
653 interface_name));
