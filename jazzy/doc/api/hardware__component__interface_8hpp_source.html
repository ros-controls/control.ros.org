379 template <typename T>
380 bool set_state(
381 const StateInterface::SharedPtr & interface_handle, const T & value, bool wait_until_set)
382 {
383 if (!interface_handle)
384 {
385 throw std::runtime_error(
386 fmt::format(
387 "State interface handle is null in hardware component: {}, while calling set_state "
388 "method. This should not happen.",
389 info_.name));
390 }
391 return interface_handle->set_value(value, wait_until_set);
392 }

419 template <typename T>
420 bool get_state(
421 const StateInterface::SharedPtr & interface_handle, T & state, bool wait_until_get) const
422 {
423 if (!interface_handle)
424 {
425 throw std::runtime_error(
426 fmt::format(
427 "State interface handle is null in hardware component: {}, while calling get_state "
428 "method. This should not happen.",
429 info_.name));
430 }
431 const bool success = interface_handle->get_value(state, wait_until_get);
432 if (!success && wait_until_get)
433 {
434 throw std::runtime_error(
435 fmt::format(
436 "Failed to get state value from interface: {} in hardware component: {}. This should "
437 "not happen.",
438 interface_handle->get_name(), info_.name));
439 }
440 return success;
441 }
