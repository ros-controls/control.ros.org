74inline NoiseDistribution get_noise_distribution(const hardware_interface::ComponentInfo& sensor)
75{
76 const std::string value = get_sensor_parameter_or(sensor, "noise_distribution", "gaussian");
77 if (value != "gaussian" && value != "uniform")
78 {
79 RCLCPP_WARN(rclcpp::get_logger("mujoco_ros2_control"),
80 "Only noise distributions of 'gaussian' or 'uniform' are allowed, but you selected '%s'. "
81 "Defaulting to 'gaussian'.",
82 value.c_str());
83 }
84 return value == "uniform" ? NoiseDistribution::kUniform : NoiseDistribution::kGaussian;
85}

93inline double get_sensor_update_period(const hardware_interface::ComponentInfo& sensor, double default_period = 0.0)
94{
95 if (sensor.parameters.count("update_rate") == 0)
96 {
97 return default_period;
98 }
99 const std::string& value = sensor.parameters.at("update_rate");
100 double rate = 0.0;
101 try
102 {
103 rate = std::stod(value);
104 }
105 catch (const std::exception&)
106 {
107 }
108 if (!(rate > 0.0) || !std::isfinite(rate))
109 {
110 RCLCPP_WARN(rclcpp::get_logger("mujoco_ros2_control"),
111 "Sensor '%s' has an invalid update_rate '%s', it must be a positive number in Hz. "
112 "Falling back to the MJCF sensor interval, or the simulation rate if unset.",
113 sensor.name.c_str(), value.c_str());
114 return default_period;
115 }
116 return 1.0 / rate;
117}

157template <int N>
158inline Eigen::Matrix<double, N, 1> sample_noise(double stddev, NoiseDistribution distribution, std::mt19937& rng)
159{
160 Eigen::Matrix<double, N, 1> samples;
161 if (distribution == NoiseDistribution::kUniform)
162 {
163 const double bound = stddev * std::sqrt(3.0);
164 std::uniform_real_distribution<double> dist(-bound, bound);
165 for (int i = 0; i < N; ++i)
166 {
167 samples[i] = dist(rng);
168 }
169 }
170 else
171 {
172 std::normal_distribution<double> dist(0.0, stddev);
173 for (int i = 0; i < N; ++i)
174 {
175 samples[i] = dist(rng);
176 }
177 }
178 return samples;
179}

203inline void add_sensor_noise(Eigen::Quaterniond& value, double stddev, NoiseDistribution distribution, std::mt19937& rng)
204{
