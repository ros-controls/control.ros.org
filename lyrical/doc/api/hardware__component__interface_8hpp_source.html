323 template <typename T>
324 bool set_state(
325 const StateInterface::SharedPtr & interface_handle, const T & value, bool wait_until_set)
326 {
327 if (!interface_handle)

331 "State interface handle is null in hardware component: {}, while calling set_state "

333 info_.name));

363 template <typename T>
364 bool get_state(
365 const StateInterface::SharedPtr & interface_handle, T & state, bool wait_until_get) const
366 {
367 if (!interface_handle)
368 {
369 throw std::runtime_error(
370 fmt::format(
371 "State interface handle is null in hardware component: {}, while calling get_state "
372 "method. This should not happen.",
373 info_.name));
374 }
375 const bool success = interface_handle->get_value(state, wait_until_get);
376 if (!success && wait_until_get)
377 {
378 throw std::runtime_error(
379 fmt::format(
380 "Failed to get state value from interface: {} in hardware component: {}. This should "
381 "not happen.",
382 interface_handle->get_name(), info_.name));
383 }
384 return success;
385 }
