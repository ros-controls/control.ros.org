317 template <typename T>

321 if (!interface_handle)

357 template <typename T>
358 bool get_state(
359 const StateInterface::SharedPtr & interface_handle, T & state, bool wait_until_get) const
360 {
361 if (!interface_handle)
362 {
363 throw std::runtime_error(
364 fmt::format(
365 "State interface handle is null in hardware component: {}, while calling get_state "
366 "method. This should not happen.",
367 info_.name));
368 }
369 const bool success = interface_handle->get_value(state, wait_until_get);
370 if (!success && wait_until_get)
371 {
372 throw std::runtime_error(
373 fmt::format(
374 "Failed to get state value from interface: {} in hardware component: {}. This should "
375 "not happen.",
376 interface_handle->get_name(), info_.name));
377 }
378 return success;
379 }
