135 void set_params(double sampling_frequency, double damping_frequency, double damping_intensity)
136 {
137 a1_ = exp(
138 -1.0 / sampling_frequency * (2.0 * M_PI * damping_frequency) /
139 (pow(10.0, damping_intensity / -10.0)));
140 b1_ = 1.0 - a1_;
141 };

204template <typename T>
205bool LowPassFilter<T>::update(const T & data_in, T & data_out)
206{
207 if (!configured_)
208 {
209 throw std::runtime_error("Filter is not configured");
210 }
211 // If this is the first call to update initialize the filter at the current state
212 // so that we dont apply an impulse to the data.
213 if (Traits::is_nan(filtered_value_) || Traits::is_empty(filtered_value_))
214 {
215 if (!Traits::is_finite(data_in))
216 {
217 return false;
218 }
219
220 initialize_state(data_in);
221 }
222 else
223 {
224 // Generic validation for all types
225 Traits::validate_input(data_in, filtered_value_, data_out);
226 }
227
228 // Filter
229 filtered_value_ = old_value_ * b1_ + filtered_old_value_ * a1_;
230 filtered_old_value_ = filtered_value_;
231
232 Traits::assign(old_value_, data_in);
233 Traits::assign(data_out, filtered_value_);
234
235 if (Traits::is_finite(data_in))
236 {
237 Traits::assign(old_value_, data_in);
238 }
239
240 Traits::add_metadata(data_out, data_in);
241
242 return true;
243}
