88 template <typename T = double>
89 [[nodiscard]] std::optional<T> get_optional(unsigned int max_tries = 10) const
90 {
91 unsigned int nr_tries = 0;
92 do
93 {
94 ++get_value_statistics_.total_counter;
95 const std::optional<T> data = state_interface_.get_optional<T>();
96 if (data.has_value())
97 {
98 return data;
99 }
100 ++get_value_statistics_.failed_counter;
101 ++nr_tries;
102 std::this_thread::yield();
103 } while (nr_tries < max_tries);
104
105 ++get_value_statistics_.timeout_counter;
106 return std::nullopt;
107 }
