54constexpr auto is_ptr_or_smart_ptr = rcpputils::is_pointer<T>::value;

64template <class T, typename mutex_type = DEFAULT_MUTEX>
65class RealtimeThreadSafeBoxBase
66{
67 static_assert(std::is_copy_constructible_v<T>, "Passed type must be copy constructible");
68
69public:
70 using mutex_t = mutex_type;
71 using type = T;
72 // Provide various constructors
73 constexpr explicit RealtimeThreadSafeBoxBase(const T & init = T{}) : value_(init) {}
74 constexpr explicit RealtimeThreadSafeBoxBase(const T && init) : value_(std::move(init)) {}
75
76 // Copy constructor
77 constexpr RealtimeThreadSafeBoxBase(const RealtimeThreadSafeBoxBase & o)
78 {
79 // Lock the other box mutex
80 std::unique_lock<mutex_t> lock(o.lock_);
81 // We do not need to lock our own mutex because we are currently in the process of being created
82 value_ = o.value_;
83 }
84
85 // Copy assignment constructor
86 constexpr RealtimeThreadSafeBoxBase & operator=(const RealtimeThreadSafeBoxBase & o)
87 {
88 // Check for self assignment (and a potential deadlock)
89 if (&o != this) {
90 // Lock the other box mutex
91 std::unique_lock<mutex_t> lock_other(o.lock_);
92 std::unique_lock<mutex_t> lock_self(lock_);
93
94 value_ = o.value_;
95 }
96 return *this;
97 }
98
99 constexpr RealtimeThreadSafeBoxBase(RealtimeThreadSafeBoxBase && o)
100 {
101 // Lock the other box mutex
102 std::unique_lock<mutex_t> lock(o.lock_);
103 // We do not need to lock our own mutex because we are currently in the process of being created
104 value_ = std::move(o.value_);
105 }
106

108 template <typename U = T>
109 constexpr RealtimeThreadSafeBoxBase(
110 const std::initializer_list<U> & init,
111 std::enable_if_t<std::is_constructible_v<U, std::initializer_list<U>>>)
112 : value_(init)
113 {
114 }

134 template <typename U = T>
135 typename std::enable_if_t<!is_ptr_or_smart_ptr<U>, bool> try_set(const T & value)
136 {
137 std::unique_lock<mutex_t> guard(lock_, std::defer_lock);
138 if (!guard.try_lock()) {
139 return false;
140 }
141 value_ = value;
142 return true;
143 }
144
150 bool try_set(const std::function<void(T &)> & func)
151 {
152 std::unique_lock<mutex_t> guard(lock_, std::defer_lock);
153 if (!guard.try_lock()) {
154 return false;
155 }
156
157 func(value_);
158 return true;
159 }

165 template <typename U = T>
166 [[nodiscard]] typename std::enable_if_t<!is_ptr_or_smart_ptr<U>, std::optional<U>> try_get() const
167 {
168 std::unique_lock<mutex_t> guard(lock_, std::defer_lock);
169 if (!guard.try_lock()) {
170 return std::nullopt;
171 }
172 return value_;
173 }
174
180 bool try_get(const std::function<void(const T &)> & func)
181 {
182 std::unique_lock<mutex_t> guard(lock_, std::defer_lock);
183 if (!guard.try_lock()) {
184 return false;
185 }
186
187 func(value_);
188 return true;
189 }

196 template <typename U = T>
197 typename std::enable_if_t<!is_ptr_or_smart_ptr<U>, void> set(const T & value)
198 {
199 std::lock_guard<mutex_t> guard(lock_);
200 // cppcheck-suppress missingReturn
201 value_ = value;
202 }
203
207 void set(const std::function<void(T &)> & func)
208 {
209 std::lock_guard<mutex_t> guard(lock_);
210 if (!func) {
211 if constexpr (is_ptr_or_smart_ptr<T>) {
212 value_ = nullptr;
213 return;
214 }
215 }
216 func(value_);
217 }

240 }

257 template <typename U = T>

259 {
260 set(value);
261 }

267 template <typename U = T, typename = typename std::enable_if_t<!is_ptr_or_smart_ptr<U>>>
268 [[nodiscard]] operator T() const
269 {
270 // Only makes sense with the getNonRT method otherwise we would return an std::optional
271 return get();
272 }
273
278 template <typename U = T, typename = typename std::enable_if_t<!is_ptr_or_smart_ptr<U>>>
279 [[nodiscard]] operator std::optional<T>() const
280 {
281 return try_get();
282 }
283
284 // In case one wants to actually use a pointer
285 // in this implementation we allow accessing the lock directly.
286 // Note: Be careful with lock.unlock().
287 // It may only be called from the thread that locked the mutex!
288 [[nodiscard]] const mutex_t & get_mutex() const { return lock_; }
289 [[nodiscard]] mutex_t & get_mutex() { return lock_; }
290
291private:
292 T value_;
293
294 // Protects access to the thing in the box. This mutex is
295 // guaranteed to be locked for no longer than the duration of the
296 // copy, so as long as the copy is realtime safe and the OS has
297 // priority inheritance for mutexes, this lock can be safely locked
298 // from within realtime.
299 mutable mutex_t lock_;
300};
