107template <typename StateCodec, typename CommandCodec>
108class ZenbeddedClient : public ZenbeddedClientBase
109{
110public:
111 using StateInitParams = typename StateCodec::InitParams;
112 using CommandInitParams = typename CommandCodec::InitParams;
113 using StateValue = typename StateCodec::Value;
114 using CommandValue = typename CommandCodec::Value;
115
116 ZenbeddedClient()
117 {
118 static_assert(CheckCodec<StateCodec>::is_valid_state_codec, "Invalid StateCodec Type!");
119 static_assert(CheckCodec<CommandCodec>::is_valid_command_codec, "Invalid CommandCodec Type!");
120 }
121
122 int init(
123 uint32_t control_freq, const StateInitParams & state_params = {},
124 const CommandInitParams & cmd_params = {})
125 {
126 if (initialized_)
127 {
128 return 0;
129 }
130
131 reset_buffers();
132
133 // Initialize state headers/codecs directly in double-buffer slots
134 StateCodec::init(state_ctx_, state_params, get_state_buffer_slot(0));
135 StateCodec::init(state_ctx_, state_params, get_state_buffer_slot(1));
136 size_t st_size = StateCodec::payload_size(state_ctx_);
137
138 // Determine command payload size using scratchpad space
139 uint8_t cmd_layout_scratch[CONFIG_ZENBEDDED_MAX_CMD_BUFFER_SIZE];
140 CommandCodec::init(cmd_ctx_, cmd_params, cmd_layout_scratch);
141 size_t cmd_size = CommandCodec::payload_size(cmd_ctx_);
142
143 return init_base(control_freq, st_size, cmd_size);
144 }
145
146 void write_state(const StateValue & value)
147 {
148 if (!initialized_)
149 {
150 return;
151 }
152 int write_idx = 0;
153 uint8_t * buf = prepare_state_write_slot(write_idx);
154 StateCodec::write(state_ctx_, value, buf);
155 commit_state_write_slot(write_idx);
156 }
157
158 bool read_command(CommandValue & out)
159 {
160 if (!initialized_)
161 {
162 return false;
163 }
164 uint8_t tmp[CONFIG_ZENBEDDED_MAX_CMD_BUFFER_SIZE];
165 if (!read_latest_command_raw(tmp, cmd_payload_size_))
166 {
167 return false;
168 }
169 return CommandCodec::read(cmd_ctx_, tmp, cmd_payload_size_, out);
170 }
171
172private:
173 typename StateCodec::Ctx state_ctx_{};
174 typename CommandCodec::Ctx cmd_ctx_{};
175};
