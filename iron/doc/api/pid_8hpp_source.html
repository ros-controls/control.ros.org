45class AntiwindupStrategy
46{
47public:
48 enum Value : int8_t
49 {
50 NONE = 0,
51 BACK_CALCULATION,
52 CONDITIONAL_INTEGRATION
53 };
54
55 constexpr AntiwindupStrategy() : value_(NONE) {}
56 constexpr AntiwindupStrategy(Value v) : value_(v) {} // NOLINT(runtime/explicit)
57
58 explicit AntiwindupStrategy(const std::string & s)
59 {
60 if (s == "back_calculation")
61 {
62 value_ = BACK_CALCULATION;
63 }
64 else if (s == "conditional_integration")
65 {
66 value_ = CONDITIONAL_INTEGRATION;
67 }
68 else
69 {
70 value_ = NONE;
71 }
72 }
73
74 operator std::string() const { return to_string(); }
75
76 constexpr bool operator==(AntiwindupStrategy other) const { return value_ == other.value_; }
77 constexpr bool operator!=(AntiwindupStrategy other) const { return value_ != other.value_; }
78
79 constexpr bool operator==(Value other) const { return value_ == other; }
80 constexpr bool operator!=(Value other) const { return value_ != other; }
81
82 std::string to_string() const
83 {
84 switch (value_)
85 {
86 case BACK_CALCULATION:
87 return "back_calculation";
88 case CONDITIONAL_INTEGRATION:
89 return "conditional_integration";
90 case NONE:
91 default:
92 return "none";
93 }
94 }
95
96private:
97 Value value_;
98};
