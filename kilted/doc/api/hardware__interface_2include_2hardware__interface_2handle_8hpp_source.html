41 int status = 0;

43 abi::__cxa_demangle(typeid(T).name(), nullptr, nullptr, &status), std::free};

242 template <typename T>
243 [[nodiscard]] bool set_value(std::unique_lock<std::shared_mutex> & lock, const T & value)
244 {
245 if (!lock.owns_lock())
246 {
247 return false;
248 }
249 // BEGIN (Handle export change): for backward compatibility
250 // TODO(Manuel) set value_ directly if old functionality is removed
251 if constexpr (std::is_same_v<T, double>)
252 {
253 // If the template is of type double, check if the value_ptr_ is not nullptr
254 THROW_ON_NULLPTR(value_ptr_);
255 *value_ptr_ = value;
256 }
257 else
258 {
259 if (!std::holds_alternative<T>(value_))
260 {
261 throw std::runtime_error(
262 fmt::format(
263 FMT_COMPILE("Invalid data type : '{}' access for interface : {} expected : '{}'"),
264 get_type_name<T>(), get_name(), data_type_.to_string()));
265 }
266 value_ = value;
267 }
268 return true;
269 // END
270 }

277 void copy(const Handle & other) noexcept
278 {
279 std::scoped_lock lock(other.handle_mutex_, handle_mutex_);
280 prefix_name_ = other.prefix_name_;
281 interface_name_ = other.interface_name_;
282 handle_name_ = other.handle_name_;
283 value_ = other.value_;
284 if (std::holds_alternative<std::monostate>(value_))
285 {
286 value_ptr_ = other.value_ptr_;
287 }
288 else
289 {
290 value_ptr_ = std::get_if<double>(&value_);
291 }
292 }

294 void swap(Handle & first, Handle & second) noexcept
295 {
296 std::scoped_lock lock(first.handle_mutex_, second.handle_mutex_);
297 std::swap(first.prefix_name_, second.prefix_name_);
298 std::swap(first.interface_name_, second.interface_name_);
299 std::swap(first.handle_name_, second.handle_name_);
300 std::swap(first.value_, second.value_);
301 std::swap(first.value_ptr_, second.value_ptr_);
302 }
