128 CallbackReturn init(
129 const HardwareInfo & hardware_info, rclcpp::Logger logger, rclcpp::Clock::SharedPtr clock)
130 {
131 system_clock_ = clock;
132 system_logger_ = logger.get_child("hardware_component.system." + hardware_info.name);
133 info_ = hardware_info;
134 if (info_.is_async)
135 {
136 RCLCPP_INFO_STREAM(
137 get_logger(), "Starting async handler with scheduler priority: " << info_.thread_priority);
138 async_handler_ = std::make_unique<realtime_tools::AsyncFunctionHandler<return_type>>();
139 async_handler_->init(
140 [this](const rclcpp::Time & time, const rclcpp::Duration & period)
141 {
142 const auto read_start_time = std::chrono::steady_clock::now();
143 const auto ret_read = read(time, period);
144 const auto read_end_time = std::chrono::steady_clock::now();
145 read_return_info_.store(ret_read, std::memory_order_release);
146 read_execution_time_.store(
147 std::chrono::duration_cast<std::chrono::nanoseconds>(read_end_time - read_start_time),
148 std::memory_order_release);
149 if (ret_read != return_type::OK)
150 {
151 return ret_read;
152 }
153 const auto write_start_time = std::chrono::steady_clock::now();
154 const auto ret_write = write(time, period);
155 const auto write_end_time = std::chrono::steady_clock::now();
156 write_return_info_.store(ret_write, std::memory_order_release);
157 write_execution_time_.store(
158 std::chrono::duration_cast<std::chrono::nanoseconds>(write_end_time - write_start_time),
159 std::memory_order_release);
160 return ret_write;
161 },
162 info_.thread_priority);
163 async_handler_->start_thread();
164 }
165 return on_init(hardware_info);
166 };

408 HardwareComponentCycleStatus trigger_read(
409 const rclcpp::Time & time, const rclcpp::Duration & period)
410 {
411 HardwareComponentCycleStatus status;
412 status.result = return_type::ERROR;
413 if (info_.is_async)
414 {
415 status.result = read_return_info_.load(std::memory_order_acquire);
416 const auto read_exec_time = read_execution_time_.load(std::memory_order_acquire);
417 if (read_exec_time.count() > 0)
418 {
419 status.execution_time = read_exec_time;
420 }
421 const auto result = async_handler_->trigger_async_callback(time, period);
422 status.successful = result.first;
423 if (!status.successful)
424 {
425 RCLCPP_WARN(
426 get_logger(),
427 "Trigger read/write called while the previous async trigger is still in progress for "
428 "hardware interface : '%s'. Failed to trigger read/write cycle!",
429 info_.name.c_str());
430 status.result = return_type::OK;
431 return status;
432 }
433 }
434 else
435 {
436 const auto start_time = std::chrono::steady_clock::now();
437 status.successful = true;
438 status.result = read(time, period);
439 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
440 std::chrono::steady_clock::now() - start_time);
441 }
442 return status;
443 }

467 HardwareComponentCycleStatus trigger_write(
468 const rclcpp::Time & time, const rclcpp::Duration & period)
469 {
470 HardwareComponentCycleStatus status;
471 status.result = return_type::ERROR;
472 if (info_.is_async)
473 {
474 status.successful = true;
475 const auto write_exec_time = write_execution_time_.load(std::memory_order_acquire);
476 if (write_exec_time.count() > 0)
477 {
478 status.execution_time = write_exec_time;
479 }
480 status.result = write_return_info_.load(std::memory_order_acquire);
481 }
482 else
483 {
484 const auto start_time = std::chrono::steady_clock::now();
485 status.successful = true;
486 status.result = write(time, period);
487 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
488 std::chrono::steady_clock::now() - start_time);
489 }
490 return status;

531 template <typename T>
532 void set_state(const std::string & interface_name, const T & value)
533 {
534 auto it = system_states_.find(interface_name);
535 if (it == system_states_.end())
536 {
537 throw std::runtime_error(
538 fmt::format(
539 FMT_COMPILE(
540 "State interface not found: {} in system hardware component: {}. "
541 "This should not happen."),
542 interface_name, info_.name));
543 }
544 auto & handle = it->second;
545 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
546 std::ignore = handle->set_value(lock, value);
547 }

549 template <typename T = double>
550 T get_state(const std::string & interface_name) const
551 {
552 auto it = system_states_.find(interface_name);
553 if (it == system_states_.end())
