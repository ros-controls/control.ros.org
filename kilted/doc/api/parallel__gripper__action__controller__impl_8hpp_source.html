30void GripperActionController::preempt_active_goal()
31{
32 // Cancels the currently active goal
33 RealtimeGoalHandlePtr active_goal;
34 rt_active_goal_.get([&](const RealtimeGoalHandlePtr & goal) { active_goal = goal; });
35 if (active_goal)
36 {
37 // Marks the current goal as canceled
38 active_goal->setCanceled(std::make_shared<GripperCommandAction::Result>());
39 rt_active_goal_.set([](RealtimeGoalHandlePtr & stored_value)
40 { stored_value = RealtimeGoalHandlePtr(); });
41 }
42}

59controller_interface::return_type GripperActionController::update(
60 const rclcpp::Time & /*time*/, const rclcpp::Duration & /*period*/)
61{
62 auto command_struct_rt_op = command_.try_get();
63 if (command_struct_rt_op.has_value())
64 {
65 command_struct_rt_ = command_struct_rt_op.value();
66 }
67
68 const double current_position = joint_position_state_interface_->get().get_value();
69 const double current_velocity = joint_velocity_state_interface_->get().get_value();
70 const double error_position = command_struct_rt_.position_cmd_ - current_position;
71
72 check_for_success(get_node()->now(), error_position, current_position, current_velocity);
73
74 joint_command_interface_->get().set_value(command_struct_rt_.position_cmd_);
75 if (speed_interface_.has_value())
76 {
77 speed_interface_->get().set_value(command_struct_rt_.max_velocity_);
78 }
79 if (effort_interface_.has_value())
80 {
81 effort_interface_->get().set_value(command_struct_rt_.max_effort_);
82 }
83
84 return controller_interface::return_type::OK;
85}

87rclcpp_action::GoalResponse GripperActionController::goal_callback(
88 const rclcpp_action::GoalUUID &, std::shared_ptr<const GripperCommandAction::Goal> goal_handle)
89{
90 if (goal_handle->command.position.size() != 1)
91 {
92 pre_alloc_result_ = std::make_shared<control_msgs::action::ParallelGripperCommand::Result>();
93 pre_alloc_result_->state.position.resize(1);
94 pre_alloc_result_->state.effort.resize(1);
95 RCLCPP_ERROR(
96 get_node()->get_logger(),
97 "Received action goal with wrong number of position values, expects 1, got %zu",
98 goal_handle->command.position.size());
99 return rclcpp_action::GoalResponse::REJECT;
100 }
101
102 RCLCPP_INFO(get_node()->get_logger(), "Received & accepted new action goal");
103 return rclcpp_action::GoalResponse::ACCEPT_AND_EXECUTE;
104}

106void GripperActionController::accepted_callback(
107 std::shared_ptr<GoalHandle> goal_handle) // Try to update goal
108{
109 auto rt_goal = std::make_shared<RealtimeGoalHandle>(goal_handle);
110
111 // Accept new goal
112 preempt_active_goal();
113
114 // This is the non-realtime command_struct
115 // We use command_ for sharing
116 command_struct_.position_cmd_ = goal_handle->get_goal()->command.position[0];
117 if (!params_.max_velocity_interface.empty() && !goal_handle->get_goal()->command.velocity.empty())
118 {
119 command_struct_.max_velocity_ = goal_handle->get_goal()->command.velocity[0];
120 }
121 else
122 {
123 command_struct_.max_velocity_ = params_.max_velocity;
124 }
125 if (!params_.max_effort_interface.empty() && !goal_handle->get_goal()->command.effort.empty())
126 {
127 command_struct_.max_effort_ = goal_handle->get_goal()->command.effort[0];
128 }
129 else
130 {
131 command_struct_.max_effort_ = params_.max_effort;
132 }
133 command_.set(command_struct_);
134
135 pre_alloc_result_->reached_goal = false;
136 pre_alloc_result_->stalled = false;
137
138 last_movement_time_ = get_node()->now();
139 rt_goal->execute();
140 rt_active_goal_.set([rt_goal](RealtimeGoalHandlePtr & stored_value) { stored_value = rt_goal; });
141
142 // Set smartpointer to expire for create_wall_timer to delete previous entry from timer list
143 goal_handle_timer_.reset();
144
145 // Setup goal status checking timer
146 goal_handle_timer_ = get_node()->create_wall_timer(
147 action_monitor_period_.to_chrono<std::chrono::nanoseconds>(),
148 std::bind(&RealtimeGoalHandle::runNonRealtime, rt_goal));
149}

151rclcpp_action::CancelResponse GripperActionController::cancel_callback(
152 const std::shared_ptr<GoalHandle> goal_handle)
153{
154 RCLCPP_INFO(get_node()->get_logger(), "Got request to cancel goal");
155
156 // Check that cancel request refers to currently active goal (if any)
157 RealtimeGoalHandlePtr active_goal;
158 rt_active_goal_.get([&](const RealtimeGoalHandlePtr & goal) { active_goal = goal; });
159 if (active_goal && active_goal->gh_ == goal_handle)
160 {
161 // Enter hold current position mode
162 set_hold_position();
163
164 RCLCPP_INFO(
165 get_node()->get_logger(), "Canceling active action goal because cancel callback received.");
166
167 // Mark the current goal as canceled
168 auto action_res = std::make_shared<GripperCommandAction::Result>();
169 active_goal->setCanceled(action_res);
170 // Reset current goal
171 rt_active_goal_.set([](RealtimeGoalHandlePtr & stored_value)
172 { stored_value = RealtimeGoalHandlePtr(); });
173 }
174 return rclcpp_action::CancelResponse::ACCEPT;
175}

185void GripperActionController::check_for_success(
186 const rclcpp::Time & time, double error_position, double current_position,
187 double current_velocity)
188{
189 RealtimeGoalHandlePtr active_goal;
190 rt_active_goal_.get([&](const RealtimeGoalHandlePtr & goal) { active_goal = goal; });
191 if (!active_goal)
192 {
193 return;
194 }
195
196 if (fabs(error_position) < params_.goal_tolerance)
197 {
198 pre_alloc_result_->state.effort[0] = computed_command_;
199 pre_alloc_result_->state.position[0] = current_position;
200 pre_alloc_result_->reached_goal = true;
201 pre_alloc_result_->stalled = false;
202 RCLCPP_DEBUG(get_node()->get_logger(), "Successfully moved to goal.");
203 active_goal->setSucceeded(pre_alloc_result_);
204 rt_active_goal_.set([](RealtimeGoalHandlePtr & stored_value)
205 { stored_value = RealtimeGoalHandlePtr(); });
206 }
207 else
208 {
209 if (fabs(current_velocity) > params_.stall_velocity_threshold)
210 {
211 last_movement_time_ = time;
212 }
213 else if ((time - last_movement_time_).seconds() > params_.stall_timeout)
214 {
215 pre_alloc_result_->state.effort[0] = computed_command_;
216 pre_alloc_result_->state.position[0] = current_position;
217 pre_alloc_result_->reached_goal = false;
218 pre_alloc_result_->stalled = true;
219
220 if (params_.allow_stalling)
221 {
222 RCLCPP_DEBUG(get_node()->get_logger(), "Stall detected moving to goal. Returning success.");
223 active_goal->setSucceeded(pre_alloc_result_);
224 }
225 else
226 {
227 RCLCPP_DEBUG(get_node()->get_logger(), "Stall detected moving to goal. Aborting action!");
228 active_goal->setAborted(pre_alloc_result_);
229 }
230 rt_active_goal_.set([](RealtimeGoalHandlePtr & stored_value)
231 { stored_value = RealtimeGoalHandlePtr(); });
232 }
233 }

236controller_interface::CallbackReturn GripperActionController::on_configure(
237 const rclcpp_lifecycle::State &)
238{
239 const auto logger = get_node()->get_logger();
240 params_ = param_listener_->get_params();
241
242 // Action status checking update rate
243 action_monitor_period_ = rclcpp::Duration::from_seconds(1.0 / params_.action_monitor_rate);
244 RCLCPP_INFO(
245 logger, "Action status changes will be monitored at %f Hz.", params_.action_monitor_rate);
246
247 // Controlled joint

249 {
250 RCLCPP_ERROR(logger, "Joint name cannot be empty");
251 return controller_interface::CallbackReturn::ERROR;

253 RCLCPP_INFO(logger, "Joint name is : %s", params_.joint.c_str());
254
255 return controller_interface::CallbackReturn::SUCCESS;

258controller_interface::CallbackReturn GripperActionController::on_activate(
259 const rclcpp_lifecycle::State &)
260{
261 auto command_interface_it = std::find_if(
262 command_interfaces_.begin(), command_interfaces_.end(),
263 [](const hardware_interface::LoanedCommandInterface & command_interface)
264 { return command_interface.get_interface_name() == hardware_interface::HW_IF_POSITION; });
265 if (command_interface_it == command_interfaces_.end())
266 {
267 RCLCPP_ERROR(
268 get_node()->get_logger(), "Expected 1 %s command interface",
269 hardware_interface::HW_IF_POSITION);
270 return controller_interface::CallbackReturn::ERROR;
271 }
272 if (command_interface_it->get_prefix_name() != params_.joint)
273 {
274 RCLCPP_ERROR(
275 get_node()->get_logger(), "Command interface is different than joint name `%s` != `%s`",
276 command_interface_it->get_prefix_name().c_str(), params_.joint.c_str());
277 return controller_interface::CallbackReturn::ERROR;
278 }
279 const auto position_state_interface_it = std::find_if(
280 state_interfaces_.begin(), state_interfaces_.end(),
281 [](const hardware_interface::LoanedStateInterface & state_interface)
282 { return state_interface.get_interface_name() == hardware_interface::HW_IF_POSITION; });
283 if (position_state_interface_it == state_interfaces_.end())
284 {
285 RCLCPP_ERROR(get_node()->get_logger(), "Expected 1 position state interface");
286 return controller_interface::CallbackReturn::ERROR;
287 }
288 if (position_state_interface_it->get_prefix_name() != params_.joint)
289 {
290 RCLCPP_ERROR(
291 get_node()->get_logger(),
292 "Position state interface is different than joint name `%s` != `%s`",
293 position_state_interface_it->get_prefix_name().c_str(), params_.joint.c_str());
294 return controller_interface::CallbackReturn::ERROR;
295 }
296 const auto velocity_state_interface_it = std::find_if(
297 state_interfaces_.begin(), state_interfaces_.end(),
298 [](const hardware_interface::LoanedStateInterface & state_interface)
299 { return state_interface.get_interface_name() == hardware_interface::HW_IF_VELOCITY; });
300 if (velocity_state_interface_it == state_interfaces_.end())
301 {
302 RCLCPP_ERROR(get_node()->get_logger(), "Expected 1 velocity state interface");
303 return controller_interface::CallbackReturn::ERROR;
304 }
305 if (velocity_state_interface_it->get_prefix_name() != params_.joint)
306 {
307 RCLCPP_ERROR(
308 get_node()->get_logger(),
309 "Velocity command interface is different than joint name `%s` != `%s`",
310 velocity_state_interface_it->get_prefix_name().c_str(), params_.joint.c_str());
311 return controller_interface::CallbackReturn::ERROR;
312 }
313
314 joint_command_interface_ = *command_interface_it;
315 joint_position_state_interface_ = *position_state_interface_it;
316 joint_velocity_state_interface_ = *velocity_state_interface_it;
317
318 for (auto & interface : command_interfaces_)
319 {
320 if (interface.get_interface_name() == "set_gripper_max_effort")
321 {
322 effort_interface_ = interface;
323 }
324 else if (interface.get_interface_name() == "set_gripper_max_velocity")
325 {
326 speed_interface_ = interface;
327 }
328 }
329
330 // Command
331 command_struct_.position_cmd_ = joint_position_state_interface_->get().get_value();
332 command_struct_.max_effort_ = params_.max_effort;
333 command_struct_.max_velocity_ = params_.max_velocity;
334 command_.try_set(command_struct_);
335
336 // Result
337 pre_alloc_result_ = std::make_shared<control_msgs::action::ParallelGripperCommand::Result>();
338 pre_alloc_result_->state.position.resize(1);
339 pre_alloc_result_->state.effort.resize(1);
340 pre_alloc_result_->state.position[0] = command_struct_.position_cmd_;
341 pre_alloc_result_->reached_goal = false;
342 pre_alloc_result_->stalled = false;
343
344 // Action interface
345 action_server_ = rclcpp_action::create_server<control_msgs::action::ParallelGripperCommand>(
346 get_node(), "~/gripper_cmd",
347 std::bind(
348 &GripperActionController::goal_callback, this, std::placeholders::_1, std::placeholders::_2),
349 std::bind(&GripperActionController::cancel_callback, this, std::placeholders::_1),
350 std::bind(&GripperActionController::accepted_callback, this, std::placeholders::_1));
351
352 return controller_interface::CallbackReturn::SUCCESS;
353}
