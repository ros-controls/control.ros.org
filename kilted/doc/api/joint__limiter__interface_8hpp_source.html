59 virtual bool init(
60 const std::vector<std::string> & joint_names,
61 const rclcpp::node_interfaces::NodeParametersInterface::SharedPtr & param_itf,
62 const rclcpp::node_interfaces::NodeLoggingInterface::SharedPtr & logging_itf,
63 const std::string & robot_description_topic = "/robot_description")
64 {
65 number_of_joints_ = joint_names.size();
66 joint_names_ = joint_names;
67 joint_limits_.resize(number_of_joints_);
68 node_param_itf_ = param_itf;
69 node_logging_itf_ = logging_itf;
70
71 bool result = true;
72
73 // TODO(destogl): get limits from URDF
74
75 // Initialize and get joint limits from parameter server
76 if (has_parameter_interface())
77 {
78 for (size_t i = 0; i < number_of_joints_; ++i)
79 {
80 if (!declare_parameters(joint_names[i], node_param_itf_, node_logging_itf_))
81 {
82 RCLCPP_ERROR(
83 node_logging_itf_->get_logger(),
84 "JointLimiter: Joint '%s': parameter declaration has failed", joint_names[i].c_str());
85 result = false;
86 break;
87 }
88 if (!get_joint_limits(joint_names[i], node_param_itf_, node_logging_itf_, joint_limits_[i]))
89 {
90 RCLCPP_ERROR(
91 node_logging_itf_->get_logger(),
92 "JointLimiter: Joint '%s': getting parameters has failed", joint_names[i].c_str());
93 result = false;
94 break;
95 }
96 RCLCPP_INFO(
97 node_logging_itf_->get_logger(), "Limits for joint %zu (%s) are:\n%s", i,
98 joint_names[i].c_str(), joint_limits_[i].to_string().c_str());
99 }
100 updated_limits_.set(joint_limits_);
101
102 auto on_parameter_event_callback = [this](const std::vector<rclcpp::Parameter> & parameters)
103 {
104 rcl_interfaces::msg::SetParametersResult set_parameters_result;
105 set_parameters_result.successful = true;
106
107 std::vector<joint_limits::JointLimits> updated_joint_limits = joint_limits_;
108 bool changed = false;
109
110 for (size_t i = 0; i < number_of_joints_; ++i)
111 {
112 changed |= joint_limits::check_for_limits_update(
113 joint_names_[i], parameters, node_logging_itf_, updated_joint_limits[i]);
114 }
115
116 if (changed)
117 {
118 updated_limits_.set(updated_joint_limits);
119 RCLCPP_INFO(node_logging_itf_->get_logger(), "Limits are dynamically updated!");
120 }
121
122 return set_parameters_result;
123 };
124
125 parameter_callback_ =
126 node_param_itf_->add_on_set_parameters_callback(on_parameter_event_callback);
127 }
128
129 if (result)
130 {
131 result = on_init();
132 }
133
134 (void)robot_description_topic; // avoid linters output
135
136 return result;
137 }
