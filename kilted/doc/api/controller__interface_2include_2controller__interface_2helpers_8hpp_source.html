42template <typename T>
43bool get_ordered_interfaces(
44 std::vector<T> & unordered_interfaces, const std::vector<std::string> & ordered_names,
45 const std::string & interface_type, std::vector<std::reference_wrapper<T>> & ordered_interfaces)
46{
47 ordered_interfaces.reserve(ordered_names.size());
48 for (const auto & name : ordered_names)
49 {
50 for (auto & interface : unordered_interfaces)
51 {
52 if (!interface_type.empty())
53 {
54 // check case where:
55 // (<joint> == <joint> AND <interface> == <interface>) OR <joint>/<interface> == 'full name'
56 if (
57 ((name == interface.get_prefix_name()) &&
58 (interface_type == interface.get_interface_name())) ||
59 ((name + "/" + interface_type) == interface.get_name()))
60 {
61 ordered_interfaces.push_back(std::ref(interface));
62 }
63 }
64 else
65 {
66 if (name == interface.get_name())
67 {
68 ordered_interfaces.push_back(std::ref(interface));
69 }
70 }
71 }
72 }
73
74 return ordered_names.size() == ordered_interfaces.size();
75}
