42class AsyncSchedulingPolicy
43{
44public:
45 enum Value : int8_t {
46 UNKNOWN = -1,
47 SYNCHRONIZED,
48 DETACHED,
49 };
50
51 AsyncSchedulingPolicy() = default;
52 constexpr AsyncSchedulingPolicy(Value value) : value_(value) {} // NOLINT(runtime/explicit)
53 explicit AsyncSchedulingPolicy(const std::string & data_type)
54 {
55 if (data_type == "synchronized") {
56 value_ = SYNCHRONIZED;
57 } else if (data_type == "detached") {
58 value_ = DETACHED;
59 } else {
60 value_ = UNKNOWN;
61 }
62 }
63
64 operator Value() const { return value_; }
65
66 explicit operator bool() const = delete;
67
68 constexpr bool operator==(AsyncSchedulingPolicy other) const { return value_ == other.value_; }
69 constexpr bool operator!=(AsyncSchedulingPolicy other) const { return value_ != other.value_; }
70
71 constexpr bool operator==(Value other) const { return value_ == other; }
72 constexpr bool operator!=(Value other) const { return value_ != other; }
73
74 std::string to_string() const
75 {
76 switch (value_) {
77 case SYNCHRONIZED:
78 return "synchronized";
79 case DETACHED:
80 return "detached";
81 default:
82 return "unknown";
83 }
84 }
85
86 AsyncSchedulingPolicy from_string(const std::string & data_type)
87 {
88 return AsyncSchedulingPolicy(data_type);
89 }
90
91private:
92 Value value_ = UNKNOWN;
93};

97 bool validate()
98 {
99 if (thread_priority < 0 || thread_priority > 99) {
100 RCLCPP_ERROR(
101 logger, "Invalid thread priority: %d. It should be between 0 and 99.", thread_priority);
102 return false;
103 }
104 if (scheduling_policy == AsyncSchedulingPolicy::DETACHED) {
105 if (!clock) {
106 RCLCPP_ERROR(logger, "Clock must be set when using DETACHED scheduling policy.");

120 RCLCPP_ERROR(logger, "The parsed trigger predicate is not valid!");
121 return false;
122 }

124 }

238 std::pair<bool, T> trigger_async_callback(
239 const rclcpp::Time & time, const rclcpp::Duration & period)
240 {
241 if (!is_initialized()) {
242 throw std::runtime_error("AsyncFunctionHandler: need to be initialized first!");
243 }
244 if (async_exception_ptr_) {
245 RCLCPP_ERROR(
246 params_.logger, "AsyncFunctionHandler: Exception caught in the async callback thread!");
247 std::rethrow_exception(async_exception_ptr_);
248 }
249 if (params_.scheduling_policy == AsyncSchedulingPolicy::DETACHED) {
250 RCLCPP_WARN_ONCE(
251 params_.logger,
252 "AsyncFunctionHandler is configured with DETACHED scheduling policy. "
253 "This means that the async callback will not be synchronized with the main thread. ");
254 if (pause_thread_.load(std::memory_order_relaxed)) {
255 {
256 std::unique_lock<std::mutex> lock(async_mtx_);
257 pause_thread_ = false;
258 RCLCPP_INFO(params_.logger, "AsyncFunctionHandler: Resuming the async callback thread.");
259 async_callback_return_ = T();
260 auto const sync_period = std::chrono::nanoseconds(1'000'000'000 / params_.exec_rate);
261 previous_time_ = params_.clock->now() - rclcpp::Duration(sync_period);
262 }
263 async_callback_condition_.notify_one();
264 }
265 return std::make_pair(true, async_callback_return_.load(std::memory_order_relaxed));
266 }
267 if (!is_running()) {
268 throw std::runtime_error(
269 "AsyncFunctionHandler: need to start the async callback thread first before triggering!");
270 }
271 std::unique_lock<std::mutex> lock(async_mtx_, std::try_to_lock);
272 bool trigger_status = false;
273 if (lock.owns_lock() && !trigger_in_progress_ && trigger_predicate_()) {
274 {
275 std::unique_lock<std::mutex> scoped_lock(std::move(lock));
276 trigger_in_progress_ = true;
277 current_callback_time_ = time;
278 current_callback_period_ = period;
279 }
280 async_callback_condition_.notify_one();
281 trigger_status = true;
282 }
283 const T return_value = async_callback_return_;
284 return std::make_pair(trigger_status, return_value);
285 }

450 void start_thread()
451 {
452 if (!is_initialized()) {
453 throw std::runtime_error("AsyncFunctionHandler: need to be initialized first!");
454 }
455 if (!thread_.joinable()) {
456 reset_variables();
457 thread_ = std::thread([this]() -> void {
458 if (!realtime_tools::configure_sched_fifo(thread_priority_)) {
459 RCLCPP_WARN(
460 params_.logger,
461 "Could not enable FIFO RT scheduling policy. Consider setting up your user to do FIFO "
462 "RT scheduling. See "
463 "[https://control.ros.org/master/doc/ros2_control/controller_manager/doc/userdoc.html] "
464 "for details.");
465 }
466 if (!params_.cpu_affinity_cores.empty()) {
467 const auto affinity_result =
468 realtime_tools::set_current_thread_affinity(params_.cpu_affinity_cores);
469 RCLCPP_WARN_EXPRESSION(
470 params_.logger, !affinity_result.first,
471 "Could not set CPU affinity for the async worker thread. Error: %s",
472 affinity_result.second.c_str());
473 RCLCPP_WARN_EXPRESSION(
474 params_.logger, affinity_result.first,
475 "Async worker thread is successfully pinned to the requested CPU cores!");
476 }
477 if (params_.scheduling_policy == AsyncSchedulingPolicy::SYNCHRONIZED) {
478 execute_synchronized_callback();
479 } else {
480 execute_detached_callback();
481 }
482 });
483 }
484 }

486private:
487 void execute_synchronized_callback()
488 {
489 while (!stop_async_callback_.load(std::memory_order_relaxed)) {
490 {
491 std::unique_lock<std::mutex> lock(async_mtx_);
492 async_callback_condition_.wait(
493 lock, [this] { return trigger_in_progress_ || stop_async_callback_; });
494 if (!stop_async_callback_) {
495 const auto start_time = std::chrono::steady_clock::now();
496 try {
497 async_callback_return_ =
498 async_function_(current_callback_time_, current_callback_period_);
499 } catch (...) {
500 async_exception_ptr_ = std::current_exception();
501 }
502 const auto end_time = std::chrono::steady_clock::now();
503 last_execution_time_ =
504 std::chrono::duration_cast<std::chrono::nanoseconds>(end_time - start_time);
505 }
506 trigger_in_progress_ = false;
507 }
508 cycle_end_condition_.notify_all();
509 }

512 void execute_detached_callback()
513 {
514 if (!params_.clock) {
515 throw std::runtime_error(
516 "AsyncFunctionHandler: Clock must be set when using DETACHED scheduling policy.");
517 }
518 if (params_.exec_rate == 0u) {
519 throw std::runtime_error(
520 "AsyncFunctionHandler: Execution rate must be set when using DETACHED scheduling policy.");
521 }
522
523 auto const period = std::chrono::nanoseconds(1'000'000'000 / params_.exec_rate);
524
525 if (pause_thread_) {
526 std::unique_lock<std::mutex> lock(async_mtx_);
527 async_callback_condition_.wait(
528 lock, [this] { return !pause_thread_ || stop_async_callback_; });
529 }
530 // for calculating the measured period of the loop
531 previous_time_ = params_.clock->now();
532 std::this_thread::sleep_for(period);
533 std::chrono::steady_clock::time_point next_iteration_time{std::chrono::steady_clock::now()};
534 while (!stop_async_callback_.load(std::memory_order_relaxed)) {
535 {
536 std::unique_lock<std::mutex> lock(async_mtx_);
537 async_callback_condition_.wait(
538 lock, [this] { return !pause_thread_ || stop_async_callback_; });
539 if (!stop_async_callback_) {
540 // calculate measured period
541 auto const current_time = params_.clock->now();
542 auto const measured_period = current_time - previous_time_;
543 previous_time_ = current_time;
544 current_callback_time_ = current_time;
545 current_callback_period_ = measured_period;
546
547 const auto start_time = std::chrono::steady_clock::now();
548 try {
549 async_callback_return_ = async_function_(current_time, measured_period);
550 } catch (...) {
551 async_exception_ptr_ = std::current_exception();
552 }
553 last_execution_time_ = std::chrono::duration_cast<std::chrono::nanoseconds>(
554 std::chrono::steady_clock::now() - start_time);
555
556 next_iteration_time += period;
557 const auto time_now = std::chrono::steady_clock::now();
558 if (next_iteration_time < time_now) {
559 const double time_diff =
560 std::chrono::duration<double, std::milli>(time_now - next_iteration_time).count();
561 const double cm_period = 1.e3 / static_cast<double>(params_.exec_rate);
562 const int overrun_count = static_cast<int>(std::ceil(time_diff / cm_period));
563 RCLCPP_WARN_THROTTLE(
564 params_.logger, *params_.clock, 1000,
565 "Overrun detected! The async callback missed its desired rate of %d Hz. The loop "
566 "took %f ms (missed cycles : %d).",
567 params_.exec_rate, time_diff + cm_period, overrun_count + 1);
568 next_iteration_time += (overrun_count * period);
569 }
570 std::this_thread::sleep_until(next_iteration_time);
571 }
572 trigger_in_progress_ = false;
573 }
574 cycle_end_condition_.notify_all();
575 }
576 }
