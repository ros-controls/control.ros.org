221 template <typename U>
222 [[nodiscard]] std::enable_if_t<std::is_convertible_v<const U &, T>, bool> bounded_push(
223 const U & data)
224 {
225 if (!push(data)) {
226 T dummy;
227 data_queue_.pop(dummy);
228 return push(data);
229 }
230 return true;
231 }
232
245 [[nodiscard]] bool get_latest(T & data)
246 {
247 if constexpr (detail::is_spsc_queue<LockFreeContainer>::value) {
248 // Snapshot the count so the work is bounded even if the producer keeps pushing meanwhile.
249 // The snapshot is a lower bound (the producer only adds), so the last element of the
250 // snapshot is still there to be popped below.
251 const std::size_t available = data_queue_.read_available();
252 for (std::size_t i = 1; i < available; ++i) {
253 data_queue_.consume_one([](const T &) {});
254 }
255 return data_queue_.pop(data);
256 } else {
257 return data_queue_.consume_all([&data](const T & d) { data = d; }) > 0;
258 }
259 }
260
268 bool empty() const
269 {
270 if constexpr (detail::is_spsc_queue<LockFreeContainer>::value) {
271 return data_queue_.read_available() == 0;
272 } else {
273 return data_queue_.empty();
274 }
275 }
