43template <typename T, std::size_t Capacity>
44class LockFreePooledSPSCQueue
45{
46 static_assert(Capacity > 0, "Capacity must be greater than zero");
47
48public:
50 LockFreePooledSPSCQueue()
51 {
52 for (std::size_t i = 0; i < Capacity; ++i) {
53 (void)free_.push(i); // cannot fail, the free queue has room for every slot
54 }
55 }
56
64 template <typename Fill>
65 [[nodiscard]] bool produce(Fill && fill)
66 {
67 std::size_t index;
68 if (!free_.pop(index)) {
69 return false;
70 }
71 fill(slots_[index]);
72 return filled_.push(index); // cannot fail, at most Capacity indices exist
73 }
74
83 // SFINAE instead of a C++20 requires-clause so the header stays usable from C++17 code. U must be
84 // T, otherwise an explicit push<U>() / pop<U>() would bypass the nothrow check on T.
85 template <
86 typename U = T,
87 std::enable_if_t<std::is_same_v<U, T> && std::is_nothrow_move_assignable_v<U>, int> = 0>
88 [[nodiscard]] bool push(T && value)
89 {
90 return produce([&value](T & slot) { slot = std::move(value); });
91 }
92
99 template <typename Use>
100 [[nodiscard]] bool consume(Use && use)
101 {
102 std::size_t index;
103 if (!filled_.pop(index)) {
104 return false;
105 }
106 use(slots_[index]);
107 return free_.push(index); // cannot fail, at most Capacity indices exist
108 }
109
117 template <
118 typename U = T,
119 std::enable_if_t<std::is_same_v<U, T> && std::is_nothrow_move_assignable_v<U>, int> = 0>
120 [[nodiscard]] bool pop(T & out)
121 {
122 return consume([&out](T & slot) { out = std::move(slot); });
123 }
124
126 bool empty() const { return filled_.empty(); }
127
128private:
129 std::array<T, Capacity> slots_;
130 LockFreeSPSCQueue<std::size_t, Capacity> filled_;
131 LockFreeSPSCQueue<std::size_t, Capacity> free_;
132};
133} // namespace realtime_tools
