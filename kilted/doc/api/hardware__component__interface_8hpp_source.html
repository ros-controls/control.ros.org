116 CallbackReturn init(const hardware_interface::HardwareComponentParams & params)
117 {
118 clock_ = params.clock;
119 auto logger_copy = params.logger;
120 logger_ = logger_copy.get_child(
121 "hardware_component." + params.hardware_info.type + "." + params.hardware_info.name);
122 info_ = params.hardware_info;
123 if (info_.is_async)
124 {
125 RCLCPP_INFO_STREAM(
126 get_logger(), "Starting async handler with scheduler priority: " << info_.thread_priority);
127 async_handler_ = std::make_unique<realtime_tools::AsyncFunctionHandler<return_type>>();
128 async_handler_->init(
129 [this](const rclcpp::Time & time, const rclcpp::Duration & period)
130 {
131 const auto read_start_time = std::chrono::steady_clock::now();
132 const auto ret_read = read(time, period);
133 const auto read_end_time = std::chrono::steady_clock::now();
134 read_return_info_.store(ret_read, std::memory_order_release);
135 read_execution_time_.store(
136 std::chrono::duration_cast<std::chrono::nanoseconds>(read_end_time - read_start_time),
137 std::memory_order_release);
138 if (ret_read != return_type::OK)
139 {
140 return ret_read;
141 }
142 if (info_.type != "sensor")
143 {
144 const auto write_start_time = std::chrono::steady_clock::now();
145 const auto ret_write = write(time, period);
146 const auto write_end_time = std::chrono::steady_clock::now();
147 write_return_info_.store(ret_write, std::memory_order_release);
148 write_execution_time_.store(
149 std::chrono::duration_cast<std::chrono::nanoseconds>(
150 write_end_time - write_start_time),
151 std::memory_order_release);
152 return ret_write;
153 }
154 return return_type::OK;
155 },
156 info_.thread_priority);
157 async_handler_->start_thread();
158 }
159
160 if (auto locked_executor = params.executor.lock())
161 {
162 std::string node_name = params.hardware_info.name;
163 std::transform(
164 node_name.begin(), node_name.end(), node_name.begin(),
165 [](unsigned char c) { return std::tolower(c); });
166 std::replace(node_name.begin(), node_name.end(), '/', '_');
167 hardware_component_node_ = std::make_shared<rclcpp::Node>(node_name);
168 locked_executor->add_node(hardware_component_node_->get_node_base_interface());
169 }
170 else
171 {
172 RCLCPP_WARN(
173 params.logger,
174 "Executor is not available during hardware component initialization for '%s'. Skipping "
175 "node creation!",
176 params.hardware_info.name.c_str());
177 }
178
179 hardware_interface::HardwareComponentInterfaceParams interface_params;
180 interface_params.hardware_info = info_;
181 interface_params.executor = params.executor;
182 return on_init(interface_params);
183 };

191 [[deprecated("Use on_init(const HardwareComponentInterfaceParams & params) instead.")]]
192 virtual CallbackReturn on_init(const HardwareInfo & hardware_info)
193 {
194 info_ = hardware_info;
195 if (info_.type == "actuator")
196 {
197 parse_state_interface_descriptions(info_.joints, joint_state_interfaces_);
198 parse_command_interface_descriptions(info_.joints, joint_command_interfaces_);
199 }
200 else if (info_.type == "sensor")
201 {
202 parse_state_interface_descriptions(info_.joints, joint_state_interfaces_);
203 parse_state_interface_descriptions(info_.sensors, sensor_state_interfaces_);
204 }
205 else if (info_.type == "system")
206 {
207 parse_state_interface_descriptions(info_.joints, joint_state_interfaces_);
208 parse_state_interface_descriptions(info_.sensors, sensor_state_interfaces_);
209 parse_state_interface_descriptions(info_.gpios, gpio_state_interfaces_);
210 parse_command_interface_descriptions(info_.joints, joint_command_interfaces_);
211 parse_command_interface_descriptions(info_.gpios, gpio_command_interfaces_);
212 }
213 return CallbackReturn::SUCCESS;
214 };

279 virtual std::vector<StateInterface::ConstSharedPtr> on_export_state_interfaces()
280 {
281 // import the unlisted interfaces
282 std::vector<hardware_interface::InterfaceDescription> unlisted_interface_descriptions =
283 export_unlisted_state_interface_descriptions();
284
285 std::vector<StateInterface::ConstSharedPtr> state_interfaces;
286 state_interfaces.reserve(
287 unlisted_interface_descriptions.size() + joint_state_interfaces_.size() +
288 sensor_state_interfaces_.size() + gpio_state_interfaces_.size());
289
290 // add InterfaceDescriptions and create the StateInterfaces from the descriptions and add to
291 // maps.
292 for (const auto & description : unlisted_interface_descriptions)
293 {
294 auto name = description.get_name();
295 unlisted_state_interfaces_.insert(std::make_pair(name, description));
296 auto state_interface = std::make_shared<StateInterface>(description);
297 hardware_states_.insert(std::make_pair(name, state_interface));
298 unlisted_states_.push_back(state_interface);
299 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
300 }
301
302 for (const auto & [name, descr] : joint_state_interfaces_)
303 {
304 auto state_interface = std::make_shared<StateInterface>(descr);
305 hardware_states_.insert(std::make_pair(name, state_interface));
306 joint_states_.push_back(state_interface);
307 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
308 }
309 for (const auto & [name, descr] : sensor_state_interfaces_)
310 {
311 auto state_interface = std::make_shared<StateInterface>(descr);
312 hardware_states_.insert(std::make_pair(name, state_interface));
313 sensor_states_.push_back(state_interface);
314 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
315 }
316 for (const auto & [name, descr] : gpio_state_interfaces_)
317 {
318 auto state_interface = std::make_shared<StateInterface>(descr);

320 gpio_states_.push_back(state_interface);
321 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
322 }
323 return state_interfaces;
324 }

459 HardwareComponentCycleStatus trigger_read(
460 const rclcpp::Time & time, const rclcpp::Duration & period)
461 {
462 HardwareComponentCycleStatus status;
463 status.result = return_type::ERROR;
464 if (info_.is_async)
465 {
466 status.result = read_return_info_.load(std::memory_order_acquire);
467 const auto read_exec_time = read_execution_time_.load(std::memory_order_acquire);
468 if (read_exec_time.count() > 0)
469 {
470 status.execution_time = read_exec_time;
471 }
472 const auto result = async_handler_->trigger_async_callback(time, period);
473 status.successful = result.first;
474 if (!status.successful)
475 {
476 RCLCPP_WARN(
477 get_logger(),
478 "Trigger read/write called while the previous async trigger is still in progress for "
479 "hardware interface : '%s'. Failed to trigger read/write cycle!",
480 info_.name.c_str());
481 status.result = return_type::OK;
482 return status;
483 }
484 }
485 else
486 {
487 const auto start_time = std::chrono::steady_clock::now();
488 status.successful = true;
489 status.result = read(time, period);
490 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
491 std::chrono::steady_clock::now() - start_time);
492 }
493 return status;
494 }

518 HardwareComponentCycleStatus trigger_write(
519 const rclcpp::Time & time, const rclcpp::Duration & period)
520 {
521 HardwareComponentCycleStatus status;
522 status.result = return_type::ERROR;
523 if (info_.is_async)
524 {
525 status.successful = true;
526 const auto write_exec_time = write_execution_time_.load(std::memory_order_acquire);
527 if (write_exec_time.count() > 0)
528 {
529 status.execution_time = write_exec_time;
530 }
531 status.result = write_return_info_.load(std::memory_order_acquire);
532 }
533 else
534 {
535 const auto start_time = std::chrono::steady_clock::now();
536 status.successful = true;
537 status.result = write(time, period);
538 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
539 std::chrono::steady_clock::now() - start_time);
540 }
541 return status;
542 }

585 template <typename T>
586 void set_state(const std::string & interface_name, const T & value)
587 {
588 auto it = hardware_states_.find(interface_name);
589 if (it == hardware_states_.end())
590 {
591 throw std::runtime_error(
592 fmt::format(
593 FMT_COMPILE(
594 "State interface not found: {} in hardware component: {}. "
595 "This should not happen."),
596 interface_name, info_.name));
597 }
598 auto & handle = it->second;
599 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
600 std::ignore = handle->set_value(lock, value);
601 }

603 template <typename T = double>
604 T get_state(const std::string & interface_name) const
605 {
606 auto it = hardware_states_.find(interface_name);
607 if (it == hardware_states_.end())
608 {
609 throw std::runtime_error(
610 fmt::format(
611 FMT_COMPILE(
612 "State interface not found: {} in hardware component: {}. "
613 "This should not happen."),
614 interface_name, info_.name));
615 }
616 auto & handle = it->second;
617 std::shared_lock<std::shared_mutex> lock(handle->get_mutex());
618 const auto opt_value = handle->get_optional<T>(lock);
619 if (!opt_value)
620 {
621 throw std::runtime_error(
622 fmt::format(
623 FMT_COMPILE("Failed to get state value from interface: {}. This should not happen."),
624 interface_name));
625 }
626 return opt_value.value();
627 }
