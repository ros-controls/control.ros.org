385 template <typename T>
386 bool set_state(
387 const StateInterface::SharedPtr & interface_handle, const T & value, bool wait_until_set)
388 {
389 if (!interface_handle)
390 {
391 throw std::runtime_error(
392 fmt::format(
393 "State interface handle is null in hardware component: {}, while calling set_state "
394 "method. This should not happen.",
395 info_.name));
396 }
397 return interface_handle->set_value(value, wait_until_set);
398 }

425 template <typename T>
426 bool get_state(
427 const StateInterface::SharedPtr & interface_handle, T & state, bool wait_until_get) const
428 {
429 if (!interface_handle)
430 {
431 throw std::runtime_error(
432 fmt::format(
433 "State interface handle is null in hardware component: {}, while calling get_state "
434 "method. This should not happen.",
435 info_.name));
436 }
437 const bool success = interface_handle->get_value(state, wait_until_get);
438 if (!success && wait_until_get)
439 {
440 throw std::runtime_error(
441 fmt::format(
442 "Failed to get state value from interface: {} in hardware component: {}. This should "
443 "not happen.",
444 interface_handle->get_name(), info_.name));
445 }
446 return success;
447 }
