372 template <typename T>
373 bool set_state(
374 const StateInterface::SharedPtr & interface_handle, const T & value, bool wait_until_set)
375 {
376 if (!interface_handle)
377 {
378 throw std::runtime_error(
379 fmt::format(
380 "State interface handle is null in hardware component: {}, while calling set_state "
381 "method. This should not happen.",
382 info_.name));
383 }
384 return interface_handle->set_value(value, wait_until_set);
385 }

412 template <typename T>
413 bool get_state(
414 const StateInterface::SharedPtr & interface_handle, T & state, bool wait_until_get) const
415 {
416 if (!interface_handle)
417 {
418 throw std::runtime_error(
419 fmt::format(
420 "State interface handle is null in hardware component: {}, while calling get_state "
421 "method. This should not happen.",
422 info_.name));
423 }
424 const bool success = interface_handle->get_value(state, wait_until_get);
425 if (!success && wait_until_get)
426 {
427 throw std::runtime_error(
428 fmt::format(
429 "Failed to get state value from interface: {} in hardware component: {}. This should "
430 "not happen.",
431 interface_handle->get_name(), info_.name));
432 }
433 return success;
434 }
