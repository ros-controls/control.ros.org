451 // Double-buffered snapshot of mj_data_: the physics loop pays for all scene-sized copies,
452 // the consumer borrows the result via acquire_data_snapshot() and performs no copy at all.
453 // The producer (any writer holding the sim mutex) fills snapshot_write_ outside any shared
454 // lock and raises snapshot_ready_; the consumer, when the flag is up, swaps the two pointers
455 // under data_exchange_mutex_ and reads snapshot_read_ in place.
456 mjData* snapshot_write_{ nullptr };
457 mjData* snapshot_read_{ nullptr };
458 bool snapshot_ready_{ false };

464 // False until apply_control_data is first called (and cleared on reset), so that initial /

466 bool control_inputs_staged_{ false };
467
468 // Buffers to track actively applied Cartesian forces from both the plugins and the Simulate /

471 std::vector<mjtNum> xfrc_viewer_capture_; // Tracks forces from the viewer

473
474 // Guards only the snapshot pointer swap and snapshot_ready_ flag.
475 // Lock order: sim_mutex_ (if needed) is always taken before this one.
476 std::mutex data_exchange_mutex_;
477
478 // Set by acquire_data_snapshot when a consumer takes the snapshot; the physics loop only
479 // runs the expensive full refresh when this is set, so refresh bandwidth tracks consumer
