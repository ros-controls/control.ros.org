127 {

139 read_async_handler_->start_thread();

202 virtual std::vector<StateInterface::ConstSharedPtr> on_export_state_interfaces()
203 {
204 // import the unlisted interfaces
205 std::vector<hardware_interface::InterfaceDescription> unlisted_interface_descriptions =
206 export_unlisted_state_interface_descriptions();
207
208 std::vector<StateInterface::ConstSharedPtr> state_interfaces;
209 state_interfaces.reserve(
210 unlisted_interface_descriptions.size() + sensor_state_interfaces_.size() +
211 joint_state_interfaces_.size());
212
213 // add InterfaceDescriptions and create StateInterfaces from the descriptions and add to maps.
214 for (const auto & description : unlisted_interface_descriptions)
215 {
216 auto name = description.get_name();
217 unlisted_state_interfaces_.insert(std::make_pair(name, description));
218 auto state_interface = std::make_shared<StateInterface>(description);
219 sensor_states_map_.insert(std::make_pair(name, state_interface));
220 unlisted_states_.push_back(state_interface);
221 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
222 }
223
224 for (const auto & [name, descr] : sensor_state_interfaces_)
225 {
226 // TODO(Manuel) check for duplicates otherwise only the first appearance of "name" is inserted
227 auto state_interface = std::make_shared<StateInterface>(descr);
228 sensor_states_map_.insert(std::make_pair(name, state_interface));
229 sensor_states_.push_back(state_interface);
230 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
231 }
232
233 for (const auto & [name, descr] : joint_state_interfaces_)
234 {
235 auto state_interface = std::make_shared<StateInterface>(descr);
236 sensor_states_map_.insert(std::make_pair(name, state_interface));
237 joint_states_.push_back(state_interface);
238 state_interfaces.push_back(std::const_pointer_cast<const StateInterface>(state_interface));
239 }
240
241 return state_interfaces;
242 }

254 HardwareComponentCycleStatus trigger_read(
255 const rclcpp::Time & time, const rclcpp::Duration & period)
256 {
257 HardwareComponentCycleStatus status;
258 status.result = return_type::ERROR;
259 if (info_.is_async)
260 {
261 const auto result = read_async_handler_->trigger_async_callback(time, period);
262 status.successful = result.first;
263 status.result = result.second;
264 const auto execution_time = read_async_handler_->get_last_execution_time();
265 if (execution_time.count() > 0)
266 {
267 status.execution_time = execution_time;
268 }
269 if (!status.successful)
270 {
271 RCLCPP_WARN(
272 get_logger(),
273 "Trigger read called while read async handler is still in progress for hardware "
274 "interface : '%s'. Failed to trigger read cycle!",
275 info_.name.c_str());
276 status.result = return_type::OK;
277 return status;
278 }
279 }
280 else
281 {
282 const auto start_time = std::chrono::steady_clock::now();
283 status.successful = true;
284 status.result = read(time, period);
285 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
286 std::chrono::steady_clock::now() - start_time);
287 }
288 return status;
289 }

330 template <typename T>
331 void set_state(const std::string & interface_name, const T & value)
332 {
333 auto it = sensor_states_map_.find(interface_name);
334 if (it == sensor_states_map_.end())
335 {
336 throw std::runtime_error(
337 fmt::format(
338 FMT_COMPILE(
339 "State interface not found: {} in sensor hardware component: {}. "
340 "This should not happen."),
341 interface_name, info_.name));
342 }
343 auto & handle = it->second;
344 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
345 std::ignore = handle->set_value(lock, value);
346 }
347
348 template <typename T = double>
349 T get_state(const std::string & interface_name) const
350 {
351 auto it = sensor_states_map_.find(interface_name);
352 if (it == sensor_states_map_.end())
353 {
354 throw std::runtime_error(
355 fmt::format(
356 FMT_COMPILE(
357 "State interface not found: {} in sensor hardware component: {}. "
358 "This should not happen."),
359 interface_name, info_.name));
360 }
361 auto & handle = it->second;
362 std::shared_lock<std::shared_mutex> lock(handle->get_mutex());
363 const auto opt_value = handle->get_optional<T>(lock);
364 if (!opt_value)
365 {
366 throw std::runtime_error(
367 fmt::format(
368 FMT_COMPILE(
369 "Failed to get state value from interface: {}. "
370 "This should not happen."),
371 interface_name));
372 }
373 return opt_value.value();
374 }
