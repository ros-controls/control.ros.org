143 CallbackReturn init(
144 const HardwareInfo & hardware_info, rclcpp::Logger logger, rclcpp::Clock::SharedPtr clock)
145 {
146 system_clock_ = clock;
147 system_logger_ = logger.get_child("hardware_component.system." + hardware_info.name);
148 info_ = hardware_info;
149 if (info_.is_async)
150 {
151 RCLCPP_INFO_STREAM(
152 get_logger(), "Starting async handler with scheduler priority: " << info_.thread_priority);
153 async_handler_ = std::make_unique<realtime_tools::AsyncFunctionHandler<return_type>>();
154 async_handler_->init(
155 [this](const rclcpp::Time & time, const rclcpp::Duration & period)
156 {
157 const auto read_start_time = std::chrono::steady_clock::now();
158 const auto ret_read = read(time, period);
159 const auto read_end_time = std::chrono::steady_clock::now();
160 read_return_info_.store(ret_read, std::memory_order_release);
161 read_execution_time_.store(
162 std::chrono::duration_cast<std::chrono::nanoseconds>(read_end_time - read_start_time),
163 std::memory_order_release);
164 if (ret_read != return_type::OK)
165 {
166 return ret_read;
167 }
168 const auto write_start_time = std::chrono::steady_clock::now();
169 const auto ret_write = write(time, period);
170 const auto write_end_time = std::chrono::steady_clock::now();
171 write_return_info_.store(ret_write, std::memory_order_release);
172 write_execution_time_.store(
173 std::chrono::duration_cast<std::chrono::nanoseconds>(write_end_time - write_start_time),
174 std::memory_order_release);
175 return ret_write;
176 },
177 info_.thread_priority);
178 async_handler_->start_thread();
179 }
180 return on_init(hardware_info);
181 };

427 HardwareComponentCycleStatus trigger_read(
428 const rclcpp::Time & time, const rclcpp::Duration & period)
429 {
430 HardwareComponentCycleStatus status;
431 status.result = return_type::ERROR;
432 if (info_.is_async)
433 {
434 status.result = read_return_info_.load(std::memory_order_acquire);
435 const auto read_exec_time = read_execution_time_.load(std::memory_order_acquire);
436 if (read_exec_time.count() > 0)
437 {
438 status.execution_time = read_exec_time;
439 }
440 const auto result = async_handler_->trigger_async_callback(time, period);
441 status.successful = result.first;
442 if (!status.successful)
443 {
444 RCLCPP_WARN(
445 get_logger(),
446 "Trigger read/write called while the previous async trigger is still in progress for "
447 "hardware interface : '%s'. Failed to trigger read/write cycle!",
448 info_.name.c_str());
449 status.result = return_type::OK;
450 return status;
451 }
452 }
453 else
454 {
455 const auto start_time = std::chrono::steady_clock::now();
456 status.successful = true;
457 status.result = read(time, period);
458 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
459 std::chrono::steady_clock::now() - start_time);
460 }
461 return status;
462 }

486 HardwareComponentCycleStatus trigger_write(
487 const rclcpp::Time & time, const rclcpp::Duration & period)
488 {
489 HardwareComponentCycleStatus status;
490 status.result = return_type::ERROR;
491 if (info_.is_async)
492 {
493 status.successful = true;
494 const auto write_exec_time = write_execution_time_.load(std::memory_order_acquire);
495 if (write_exec_time.count() > 0)
496 {
497 status.execution_time = write_exec_time;
498 }
499 status.result = write_return_info_.load(std::memory_order_acquire);
500 }
501 else
502 {
503 const auto start_time = std::chrono::steady_clock::now();
504 status.successful = true;
505 status.result = write(time, period);
506 status.execution_time = std::chrono::duration_cast<std::chrono::nanoseconds>(
507 std::chrono::steady_clock::now() - start_time);
508 }
509 return status;
510 }

550 template <typename T>
551 void set_state(const std::string & interface_name, const T & value)
552 {
553 auto it = system_states_.find(interface_name);
554 if (it == system_states_.end())
555 {
556 throw std::runtime_error(
557 "State interface not found: " + interface_name +
558 " in system hardware component: " + info_.name + ". This should not happen.");
559 }
560 auto & handle = it->second;
561 std::unique_lock<std::shared_mutex> lock(handle->get_mutex());
562 std::ignore = handle->set_value(lock, value);
563 }

567 {
