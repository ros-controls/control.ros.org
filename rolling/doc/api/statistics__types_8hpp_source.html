180 {

182
183 current_measurement_ = item;
184 if (std::isfinite(item))
185 {
186 statistics_data_.sample_count = statistics_data_.sample_count + 1;
187 const double previous_average = statistics_data_.average;
188 statistics_data_.average =
189 previous_average + (current_measurement_ - previous_average) /
190 static_cast<double>(statistics_data_.sample_count);
191 statistics_data_.min = std::min(statistics_data_.min, current_measurement_);
192 statistics_data_.max = std::max(statistics_data_.max, current_measurement_);
193 sum_of_square_diff_from_mean_ =
194 sum_of_square_diff_from_mean_ + (current_measurement_ - previous_average) *
195 (current_measurement_ - statistics_data_.average);
196 statistics_data_.standard_deviation = std::sqrt(
197 sum_of_square_diff_from_mean_ / static_cast<double>(statistics_data_.sample_count));
198 }
199 }

242class TimeWindowStatistics
243{
244public:
245 using StatisticData = ros2_control::StatisticData;
246
248 static constexpr std::size_t NUM_BUCKETS = 10;
249
251 TimeWindowStatistics() = default;
252
256 explicit TimeWindowStatistics(double window_duration) { set_window_duration(window_duration); }
257
263 void set_window_duration(double window_duration)
264 {
265 std::lock_guard<DEFAULT_MUTEX> lock(mutex_);
266 window_duration_ = window_duration;
267 const bool enabled = std::isfinite(window_duration) && window_duration > 0.0;
268 bucket_width_ = enabled ? window_duration / static_cast<double>(NUM_BUCKETS) : 0.0;
269 reset_locked();
270 enabled_.store(enabled, std::memory_order_release);
271 }
272
274 double get_window_duration() const
275 {
276 std::lock_guard<DEFAULT_MUTEX> lock(mutex_);
277 return window_duration_;
278 }
279
281 bool is_enabled() const { return enabled_.load(std::memory_order_acquire); }
282
284 void reset()
285 {
286 std::lock_guard<DEFAULT_MUTEX> lock(mutex_);
287 reset_locked();
288 }
289
298 void add_measurement(const double item, const double time_seconds)
299 {
300 // Disabled by default: don't take the lock in the real-time loop for nothing
301 if (!is_enabled() || !std::isfinite(item) || !std::isfinite(time_seconds))
302 {
303 return;
304 }
305 std::lock_guard<DEFAULT_MUTEX> guard{mutex_};
306 if (bucket_width_ <= 0.0)
307 {
308 return; // disabled concurrently
309 }
310 const double index = std::floor(time_seconds / bucket_width_);
311 // Out of int64 range (e.g. a tiny window_duration), the cast below and the window bounds
312 // computed from the index would overflow; such a sample can't be placed in time
313 if (std::abs(index) > 1e18)
314 {
315 return;
316 }
317 const auto id = static_cast<int64_t>(index);
318 if (id < latest_id_)
319 {
320 reset_locked();
321 }
322 latest_id_ = std::max(latest_id_, id);
323
324 // A slot holding another id belongs to a bucket that has already left the window
325 Bucket & bucket = buckets_[slot(id)];
326 if (bucket.id != id)
327 {
328 bucket = Bucket{};
329 bucket.id = id;
330 }
331 // Welford's online algorithm, as in MovingAverageStatistics
332 ++bucket.count;
333 const double delta = item - bucket.mean;
334 bucket.mean += delta / static_cast<double>(bucket.count);
335 bucket.m2 += delta * (item - bucket.mean);
336 bucket.min = std::min(bucket.min, item);
337 bucket.max = std::max(bucket.max, item);
338 }
339
345 StatisticData get_statistics() const
346 {
347 // Copy under the lock and merge outside of it, to keep the real-time writer's wait short
348 std::array<Bucket, NUM_BUCKETS> buckets;
349 int64_t latest_id;
350 {
351 std::lock_guard<DEFAULT_MUTEX> lock(mutex_);
352 buckets = buckets_;
353 latest_id = latest_id_;
354 }
355
356 uint64_t count = 0;
357 double mean = 0.0;
358 double m2 = 0.0;
359 double min = std::numeric_limits<double>::max();
360 double max = std::numeric_limits<double>::lowest();
361 for (const auto & bucket : buckets)
362 {
363 const bool in_window = bucket.count > 0 && bucket.id <= latest_id &&
364 bucket.id > latest_id - static_cast<int64_t>(NUM_BUCKETS);
365 if (!in_window)
366 {
367 continue;
368 }
369 // Merge of two populations (Chan et al.), numerically stable unlike a sum of squares
370 const uint64_t merged_count = count + bucket.count;
371 const double delta = bucket.mean - mean;
372 const double weight = static_cast<double>(bucket.count) / static_cast<double>(merged_count);
373 mean += delta * weight;
374 m2 += bucket.m2 + delta * delta * static_cast<double>(count) * weight;
375 count = merged_count;
376 min = std::min(min, bucket.min);
377 max = std::max(max, bucket.max);
378 }
379 StatisticData data;
380 if (count == 0)
381 {
382 data.average = std::numeric_limits<double>::quiet_NaN();
383 data.min = std::numeric_limits<double>::quiet_NaN();
384 data.max = std::numeric_limits<double>::quiet_NaN();
385 data.standard_deviation = std::numeric_limits<double>::quiet_NaN();
386 data.sample_count = 0;
387 return data;
388 }
389 data.average = mean;
390 data.min = min;
391 data.max = max;
392 data.standard_deviation = std::sqrt(m2 / static_cast<double>(count));
393 data.sample_count = count;
394 return data;
395 }
396
397private:
398 struct Bucket
399 {
400 int64_t id = std::numeric_limits<int64_t>::min();
401 uint64_t count = 0;
402 double mean = 0.0;
403 double m2 = 0.0;
404 double min = std::numeric_limits<double>::max();
405 double max = std::numeric_limits<double>::lowest();
406 };
407
408 static std::size_t slot(int64_t id)
409 {
410 const auto n = static_cast<int64_t>(NUM_BUCKETS);
411 return static_cast<std::size_t>(((id % n) + n) % n);
412 }
413
414 void reset_locked()
415 {
416 buckets_.fill(Bucket{});
417 latest_id_ = std::numeric_limits<int64_t>::min();
418 }
419
420 mutable DEFAULT_MUTEX mutex_;
421 std::array<Bucket, NUM_BUCKETS> buckets_{};
422 double window_duration_ = -1.0;
423 double bucket_width_ = 0.0;
424 // Atomic so that the disabled (default) case is checked without locking
425 std::atomic<bool> enabled_{false};
426 int64_t latest_id_ = std::numeric_limits<int64_t>::min();
427};

440class CompletedExecutionRecord
441{
442public:
444 struct Execution
445 {
447 uint64_t sequence;
449 std::chrono::nanoseconds duration;
450 };
451
453 void publish(std::chrono::nanoseconds duration)
454 {
455 const uint64_t version = version_.load(std::memory_order_relaxed);
456 // An odd version tells the consumer that the duration is being written
457 version_.store(version + 1, std::memory_order_relaxed);
458 std::atomic_thread_fence(std::memory_order_release);
459 duration_ns_.store(duration.count(), std::memory_order_relaxed);
460 version_.store(version + 2, std::memory_order_release);
461 }
462
468 std::optional<Execution> consume_new()
469 {
470 const uint64_t version = version_.load(std::memory_order_acquire);
471 if (version == consumed_version_ || (version & 1u) != 0u)
472 {
473 return std::nullopt;
474 }
475 const auto duration_ns = duration_ns_.load(std::memory_order_relaxed);
476 std::atomic_thread_fence(std::memory_order_acquire);
477 if (version_.load(std::memory_order_relaxed) != version)
478 {
479 return std::nullopt; // overwritten while reading it
480 }
481 consumed_version_ = version;
482 return Execution{version / 2, std::chrono::nanoseconds(duration_ns)};
483 }
484
486 void discard()
487 {
488 const uint64_t version = version_.load(std::memory_order_acquire);
489 // A publish in progress completes after this call, so that execution is still new
490 consumed_version_ = version - (version & 1u);
491 }
492
493private:
494 std::atomic<uint64_t> version_{0};
495 std::atomic<int64_t> duration_ns_{0};
496 // Only accessed by the consumer thread
497 uint64_t consumed_version_ = 0;
498};
